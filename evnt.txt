import gevent
from gevent import monkey, pool
import requests
import time
import logging
import csv
from gevent.lock import Semaphore

# Apply monkey patching for gevent to work with requests
monkey.patch_all()

# Configuration
URL = "https://your-api.com/endpoint"
CONCURRENCY = 10  # Max concurrent requests
DURATION = 30  # Run test for 30 seconds

# Create a pool of green threads
request_pool = pool.Pool(CONCURRENCY)

# Counters and storage
total_requests = 0
success_count = 0
error_count = 0
responses = []
lock = Semaphore()  # Semaphore for thread safety

# Configure logging
logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")

def send_request():
    """Send an API request and log responses/errors."""
    global total_requests, success_count, error_count
    start_time = time.time()

    try:
        response = requests.get(URL, timeout=5)
        elapsed_time = time.time() - start_time

        response_data = {
            "status_code": response.status_code,
            "response_time": elapsed_time,
            "body": response.text[:500]  # Limit stored response size
        }
        responses.append(response_data)

        with lock:
            total_requests += 1
            if response.status_code == 200:
                success_count += 1
            else:
                error_count += 1

        logging.info(f"Request completed in {elapsed_time:.3f}s - Status: {response.status_code}")

    except requests.exceptions.RequestException as e:
        elapsed_time = time.time() - start_time
        response_data = {"status_code": "ERROR", "response_time": elapsed_time, "body": str(e)}
        responses.append(response_data)

        with lock:
            total_requests += 1
            error_count += 1
        
        logging.error(f"Request failed after {elapsed_time:.3f}s - Error: {str(e)}")

def run_test():
    """Run the load test while maintaining concurrency."""
    start_time = time.time()

    while time.time() - start_time < DURATION:
        request_pool.spawn(send_request)

        elapsed = time.time() - start_time
        remaining_time = max(0, (DURATION - elapsed) / (CONCURRENCY * 10))
        gevent.sleep(remaining_time)

    request_pool.join()

def save_results():
    """Save responses to a CSV file."""
    with open("load_test_results.csv", "w", newline="") as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow(["Status Code", "Response Time", "Body"])
        for res in responses:
            writer.writerow([res["status_code"], res["response_time"], res["body"]])

if __name__ == "__main__":
    print(f"Starting load test with {CONCURRENCY} concurrent requests for {DURATION} seconds...")
    run_test()
    print(f"Test completed! Total: {total_requests}, Success: {success_count}, Errors: {error_count}")
    save_results()
    print("Results saved to load_test_results.csv")
